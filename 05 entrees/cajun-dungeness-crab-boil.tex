\section*{Directions}

\noindent
Thaw \textbf{crabs} overnight in the refrigerator ---
Halve \textbf{onions} ---
Halve \textbf{garlic heads} crosswise ---
Halve \textbf{lemons} ---
Scrub \textbf{potatoes}; halve any larger than a golf ball ---
Cut \textbf{carrots} into thick chunks ---
Halve \textbf{corn} ---
Cut \textbf{bell pepper} into large pieces ---
Cut \textbf{zucchini} into thick coins ---
Rinse and split each \textbf{crab} in half through the body ---
Rinse 1\nicefrac{1}{2}~cups \textbf{rice} until the water runs clear

\begin{enumerate}
    \item Fill a 20--24~qt. pot (or canner) with 8~qt. \textbf{water}. Add \textbf{onions}, \textbf{garlic heads}, \textbf{lemons} (squeezed, then dropped in), 2~\textbf{bay leaves}, and \nicefrac{1}{4}~cup \textbf{Tony Chachere's}. Cover and bring to a rolling boil over high heat, \textit{15--25~minutes}.

    \item While the pot heats, start the rice maker: 1\nicefrac{1}{2}~cups \textbf{rice}, 1\nicefrac{1}{2}~cups \textbf{hot water}, 1\nicefrac{1}{2}~tsp. \textbf{chicken Better Than Bouillon}, 1~\textbf{bay leaf}, and 1~Tbsp. \textbf{dehydrated soup vegetables}.

    \item When the water boils hard, taste it. Add \textbf{Tony Chachere's} until it tastes too salty and too spicy. Add 1~Tbsp. \textbf{vegetable Better Than Bouillon} until it has some roundness, not just heat and salt.

    \item Add \textbf{potatoes} and \textbf{carrots}. Water should cover them by about 2~inches; add hot water as needed. After the water returns to a boil, cook \textit{8--10~minutes}, until a knife pierces the outer flesh but the center is still firm. They finish with the rest of the pot.

    \item Add \textbf{corn}. Cook \textit{4~minutes}. Potatoes should still hold their shape.

    \item Add \textbf{bell pepper}. Cook \textit{2~minutes}.

    \item Add \textbf{crab}, lid on. Cook \textit{4~minutes}.

    \item Add \textbf{zucchini}. Cook \textit{2--3~minutes}, until the \textbf{crab} body is hot when you crack a knuckle, the \textbf{potatoes} are tender with a little resistance, the \textbf{corn} is hot through, and the \textbf{zucchini} is just tender and still slightly firm. Continue in \textit{2~minute} stretches until the crab is hot. Do not soak.

    \item Lift crabs and vegetables onto a platter. Dust with \textbf{Slap Ya Mama}.

    \item Ladle 2~cups of the boil broth into a small saucepan. Reduce by about a quarter, \textit{3--5~minutes}. Lower the heat. Whisk in 8~Tbsp. \textbf{butter}, then \nicefrac{1}{4}~cup \textbf{heavy cream}. Taste. Add \textbf{Slap Ya Mama} or \textbf{lemon} until it is spicy, salty, and thick enough to coat rice.

    \item Fluff the rice, remove the \textbf{bay leaf}, and fold in \nicefrac{1}{2}~Tbsp. \textbf{butter}.

    \item Spoon rice into bowls. Add vegetables and both halves of one \textbf{crab}. Spoon the sauce over. Serve at once---the crab is at its best right now.
\end{enumerate}

\subsection*{Yield}
\begin{itemize}
    \item Serves 3 (one \textbf{crab} each)
    \item Leftover vegetables are expected
\end{itemize}

\subsection*{Equipment Required}
\begin{itemize}
    \item 20--24~qt. stockpot or water-bath canner
    \item Rice maker
    \item Fine-mesh strainer
    \item Small saucepan
    \item Tongs, ladle, spider or colander
    \item Whisk
    \item Chef's knife and cutting board
    \item Measuring cups and spoons
    \item Large platter
\end{itemize}

\subsection*{Mise en Place}
\begin{itemize}
    \item Thaw \textbf{crabs} overnight
    \item Halve \textbf{onions}, \textbf{garlic heads}, and \textbf{lemons}
    \item Scrub \textbf{potatoes}; cut \textbf{carrots}, \textbf{corn}, \textbf{bell pepper}, and \textbf{zucchini}
    \item Split each \textbf{crab} in half through the body
    \item Rinse 1\nicefrac{1}{2}~cups \textbf{rice}
    \item Have \nicefrac{1}{4}~cup \textbf{Tony Chachere's}, 1~Tbsp. \textbf{vegetable Better Than Bouillon}, 1\nicefrac{1}{2}~tsp. \textbf{chicken Better Than Bouillon}, 8\nicefrac{1}{2}~Tbsp. \textbf{butter}, and \nicefrac{1}{4}~cup \textbf{heavy cream} ready
\end{itemize}

\subsection*{Ingredient Tips}
\begin{itemize}
    \item These \textbf{crabs} are already cooked. Heat them through. A long soak makes the meat rubbery.
    \item No lemons: use \nicefrac{1}{4}~cup cider vinegar in the pot instead.
    \item Fish, lobster, or clam \textbf{Better Than Bouillon} is even better in the pot than vegetable. Do not use chicken in the boil---that belongs in the rice.
    \item \textbf{Tony Chachere's} and \textbf{Slap Ya Mama} are both salty. Tony's goes in the water. Slap Ya Mama goes on the platter.
    \item \textbf{Dehydrated soup vegetables}: garlic, onion, jalape\~{n}o, and cilantro.
    \item Large potatoes: halve them so they cook with the small ones.
\end{itemize}

\subsection*{Preparation Tips}
\begin{itemize}
    \item Width matters more than height. A crab that barely fits a dinner plate needs about 12~inches of pot diameter. Split the crabs if you are using a squat canner.
    \item Taste the water before any food goes in. Vegetables dilute it.
    \item The rice maker will likely finish before the boil. Leave it on warm. Fold in \textbf{butter} at serving.
    \item \textbf{Potatoes} keep cooking after corn, pepper, crab, and zucchini go in. Pull them to the platter as soon as they are tender, even if that is before the \textbf{crab} comes out.
    \item \textbf{Zucchini} turns to mush if it sits in the pot. Thick coins, last \textit{2--3~minutes} only.
    \item Do not put \textbf{cream} in the big pot. Only in the small saucepan of reduced broth.
\end{itemize}

\subsection*{Make Ahead \& Storage}
\begin{itemize}
    \item Thaw \textbf{crabs} in the refrigerator \textit{12--24~hours}. Cold-water backup: sealed bag, \textit{1--2~hours}, water changed every \textit{30~minutes}.
    \item Leftover vegetables keep \textit{3--4~days} refrigerated. Reheat in a splash of leftover broth.
    \item Leftover \textbf{crab} keeps \textit{1--2~days}. Steam \textit{4--5~minutes} to warm. Do not reheat twice.
    \item Sauce keeps \textit{2~days}. Warm on low; do not boil.
    \item \textbf{Rice} keeps \textit{3--4~days}. Reheat with a splash of water.
\end{itemize}

\subsection*{Serving Suggestions}
\begin{itemize}
    \item One whole \textbf{crab} (both halves) per bowl. Extra vegetables in the center of the table.
    \item Extra \textbf{Slap Ya Mama} and lemon wedges on the table.
    \item Next-day leftover vegetables and rice with the leftover sauce is lunch. The crab is a same-night dish.
\end{itemize}
