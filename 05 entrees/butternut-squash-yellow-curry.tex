\begin{quote}
\textit{Aromatic jasmine rice starts in the rice cooker, then bone-in chicken leg quarters are pressure-cooked with curry paste and Better Than Bouillon. The strained broth seasons a coconut yellow curry of butternut squash and carrots; bite-size chicken and pineapple finish the pot with brown sugar, fish sauce, and lime. Serves 8.}
\end{quote}

\section*{Directions}

\noindent
Peel, seed, and cut 1 \textbf{butternut~squash} into 1" chunks ---
Peel and large-dice 4 \textbf{carrots} into \nicefrac{3}{4}--1" pieces ---
Combine \textbf{squash} and \textbf{carrots} in \textit{Large Bowl~\#1} ---
Large-dice 2 \textbf{white~onions}; mince 8 \textbf{garlic~cloves} and 1~Tbsp. \textbf{ginger}; combine in \textit{Medium Bowl~\#1} ---
Juice 1 \textbf{lime}; wedge the other ---
Chop 1~cup \textbf{cilantro}

\subsection*{Rice}

\noindent
Wash 3 scoops \textbf{jasmine~rice} until the water runs clear; drain. Cook in the rice cooker with 3~cups \textbf{water}, \nicefrac{1}{2}~Tbsp. \textbf{coconut~oil}, 1 \textbf{bay~leaf}, and 2 \textbf{green~cardamom~pods}. Leave on warm while you make the \textbf{chicken} and curry. Before serving, rest \textit{5~minutes} if just finished, remove the 1 \textbf{bay~leaf} and 2 \textbf{cardamom~pods}, and fluff.

\newpage

\subsection*{Chicken}

\begin{enumerate}
    \item Dissolve 1~Tbsp. \textbf{Better~Than~Bouillon} and 1~Tbsp. \textbf{curry~paste} in 2~cups \textbf{hot~water}.

    \item Add 1 \textbf{bay~leaf} and 3 \textbf{green~cardamom~pods} to a 6-quart Instant Pot. Place 4 \textbf{chicken~leg~quarters} on top and pour the mixture over the \textbf{chicken}. Seal the lid and set the valve to sealing. Cook on \textit{high} pressure for \textit{12~minutes}, then allow natural release for \textit{10~minutes} before releasing any remaining pressure. \textbf{Chicken} is done when it reaches \textit{165°F} at the thickest part near the bone and the meat pulls easily from the bone.

    \item Transfer \textbf{chicken} to a cutting board and rest for about \textit{10~minutes}, until cool enough to handle. Discard the skin. Pick the meat from the bones into bite-size chunks (some shredding is natural) and place in \textit{Large Bowl~\#2}.

    \item Strain the chicken broth through a fine-mesh strainer into a bowl, then skim the fat. Measure 2~cups of the skimmed broth into \textit{Medium Bowl~\#2} for the curry. Refrigerate any remaining broth for another use.
\end{enumerate}

\subsection*{Curry}

\begin{enumerate}
    \item Heat 2~Tbsp. \textbf{coconut~oil} in a large Dutch oven over medium heat. Add \textbf{onions}, \textbf{garlic}, and \textbf{ginger} (\textit{Medium Bowl~\#1}) with \nicefrac{1}{4}~tsp. \textbf{kosher~salt} and cook, stirring often, until the \textbf{onions} are translucent and the aromatics are fragrant, about \textit{6--8~minutes}. Do not brown.

    \item Add 7~Tbsp. \textbf{Mae~Ploy~yellow~curry~paste} and cook, stirring constantly, until the paste darkens slightly and smells toasted and fragrant, about \textit{2--3~minutes}.

    \item Pour in about half of the first can of \textbf{coconut~cream} and stir until the paste is smooth. Add the remaining \textbf{coconut~cream} from both cans, 2~cups reserved broth (\textit{Medium Bowl~\#2}), 4~Tbsp. \textbf{brown~sugar}, 2~Tbsp. \textbf{fish~sauce}, and 2 bruised \textbf{lemongrass~stalks} if using. Stir and bring to a simmer.

    \item Add \textbf{squash} and \textbf{carrots} (\textit{Large Bowl~\#1}) and return to a gentle simmer. Cook uncovered, stirring occasionally, for about \textit{15~minutes} until both are tender: a paring knife inserted into the center of a chunk meets no resistance, and pieces hold their shape but are soft. The sauce should coat the back of a spoon. Continue simmering in \textit{2--3~minute} increments until the vegetables are tender.

    \item Drain the \textbf{pineapple~chunks} in the can. Add the picked \textbf{chicken} (\textit{Large Bowl~\#2}), drained \textbf{pineapple}, and 1~Tbsp. \textbf{dried~holy~basil}. Simmer for \textit{3--5~minutes} until the \textbf{chicken} is heated through and steaming hot throughout.

    \item Remove the \textbf{lemongrass} if used. Taste and adjust seasoning with \textbf{salt} if needed. Stir in the juice of 1 \textbf{lime} and \nicefrac{1}{2}~cup of the chopped \textbf{cilantro}. Serve the curry over the fluffed \textbf{rice}, garnished with the remaining \textbf{cilantro} and \textbf{lime~wedges}. The curry is best soon after finishing, while the sauce is glossy and the lime is bright.
\end{enumerate}

\subsection*{Equipment Required}
\begin{itemize}
    \item Rice cooker
    \item Instant Pot (6-quart)
    \item Large Dutch oven or heavy pot (6--7~quart capacity)
    \item Fine-mesh strainer (for washing rice and straining broth)
    \item Cutting board and chef's knife
    \item Measuring cups and spoons

    \item Tongs or slotted spoon
    \item Vegetable peeler
    \item Medium prep bowls (2)
    \item Large prep bowls (2)
\end{itemize}

\subsection*{Mise en Place}
\begin{itemize}
    \item Wash 3 scoops \textbf{jasmine~rice} first; start the rice cooker before the Instant Pot
    \item Medium Bowl \#1 --- aromatics: large-dice \textbf{onions}, minced \textbf{garlic}, minced \textbf{ginger}
    \item Large Bowl \#1 --- vegetables: 1" \textbf{butternut~squash} chunks and large-dice \textbf{carrots}
    \item Large Bowl \#2 --- picked bite-size \textbf{chicken} (after Instant Pot)
    \item Medium Bowl \#2 --- 2~cups strained, skimmed chicken broth
    \item Divide shared aromatics before cooking: \nicefrac{1}{2}~Tbsp. \textbf{coconut~oil}, 1 \textbf{bay~leaf}, and 2 \textbf{cardamom~pods} for rice; 2~Tbsp. \textbf{coconut~oil}, 1 \textbf{bay~leaf}, and 3 \textbf{cardamom~pods} for chicken/curry
    \item Juice 1 \textbf{lime} and wedge the other; chop 1~cup \textbf{cilantro} (\nicefrac{1}{2}~cup for the curry, remainder for garnish)
\end{itemize}

\subsection*{Ingredient Tips}
\begin{itemize}
    \item Mae Ploy \textbf{yellow~curry~paste} is recommended; 8~Tbsp. total is intentional and assertive---taste and adjust with \textbf{coconut~cream} or \textbf{brown~sugar} if needed
    \item \textbf{Coconut~cream} significantly improves body and richness; full-fat \textbf{coconut~milk} is an acceptable substitute but yields a thinner, less luxurious curry
    \item \textbf{Brown~sugar} (not palm) matches this tested balance with pineapple
    \item \textbf{Fish~sauce} brands vary in saltiness; start with 2~Tbsp. and adjust at the end
    \item \textbf{Dried~holy~basil} adds herbal lift when stirred in at the finish
    \item Optional \textbf{lemongrass}: bruise the stalks with the flat of a knife before adding to the curry; remove before serving
    \item Bone-in, skin-on \textbf{leg~quarters} yield richer Instant Pot broth; discard skin after cooking
\end{itemize}

\subsection*{Preparation Tips}
\begin{itemize}
    \item Cook in order: \textbf{rice} first, then Instant Pot \textbf{chicken}, then curry
    \item Wash \textbf{rice} until the water runs clear (\textit{3--4~rinses}) so grains stay separate
    \item Fully dissolve \textbf{Better~Than~Bouillon} and the 1~Tbsp. \textbf{curry~paste} in the 2~cups \textbf{hot~water} before pouring over the \textbf{chicken}
    \item While the Instant Pot cooks, finish mise en place and measure the curry seasonings
    \item Bloom the remaining 7~Tbsp. \textbf{curry~paste} until it darkens and smells toasted---raw paste tastes flat
    \item Uniform 1" \textbf{squash} and similar \textbf{carrot} pieces cook evenly; larger chunks need a few more minutes
    \item The Instant Pot cooks the \textbf{chicken} through; the final simmer only reheats it with pineapple
    \item Skim the chicken broth well so the curry is rich but not greasy
    \item Taste before serving; \textbf{lime~juice} and \textbf{salt} balance the sweet coconut sauce
\end{itemize}

\subsection*{Make Ahead \& Storage}
\begin{itemize}
    \item Curry can be made \textit{1--2~days} ahead; flavors improve
    \item Store refrigerated up to \textit{4~days}
    \item Reheat gently; thin with reserved Instant Pot broth or \textbf{coconut~cream} if the sauce has thickened
    \item Add fresh \textbf{lime~juice} and \textbf{cilantro} after reheating
    \item Extra Instant Pot broth keeps refrigerated \textit{4--5~days} or freezes well
    \item Cooked \textbf{rice} keeps refrigerated \textit{3--4~days}; reheat with a splash of water
\end{itemize}

\subsection*{Serving Suggestions}
\begin{itemize}
    \item Serve the curry over the aromatic rice-cooker \textbf{rice}
    \item \textbf{Lime~wedges} and remaining \textbf{cilantro} at the table
    \item Fresh Thai basil at the table if available
\end{itemize}
