\section*{Directions}

\subsection*{Rice}

\begin{enumerate}
    \item  Wash 3 scoops \textbf{jasmine~rice} until the water runs clear; drain.
    \item  Cook in the rice cooker with 3~cups \textbf{water}, \nicefrac{1}{2}~Tbsp. \textbf{coconut~oil}, 1 \textbf{bay~leaf}, and 2 \textbf{green~cardamom~pods}.
    \item Remove the 1 \textbf{bay~leaf} and 2 \textbf{cardamom~pods}, and fluff. Leave on warm while you make the \textbf{chicken} and curry.
\end{enumerate}

% \newpage

\subsection*{Chicken}

\begin{enumerate}
    \item Dissolve 1~Tbsp. \textbf{Better~Than~Bouillon} and 1~Tbsp. \textbf{curry~paste} in 2~cups \textbf{hot~water}.

    \item Add 1 \textbf{bay~leaf} and 3 \textbf{green~cardamom~pods} to a 6-quart Instant Pot. Place 4 \textbf{chicken~leg~quarters} on top and pour the mixture over the \textbf{chicken}. Seal the lid and set the valve to sealing. Cook on \textit{high} pressure for \textit{12~minutes}, then allow natural release for \textit{10~minutes} before releasing any remaining pressure. \textbf{Chicken} is done when it reaches \textit{165°F} at the thickest part near the bone and the meat pulls easily from the bone. \textit{While the Instant Pot comes to pressure and cooks, prep and cook the Curry through the squash-and-carrot simmer.}

    \item Transfer \textbf{chicken} to a cutting board and rest for about \textit{10~minutes}, until cool enough to handle. Discard the skin. Pick the meat from the bones into bite-size chunks (some shredding is natural) and set aside in \textit{Medium Bowl~\#1}.

    \item Strain the chicken broth through a fine-mesh strainer into a bowl, then skim the fat. Measure 2~cups of the skimmed broth into \textit{Medium Bowl~\#2}. If the curry is already simmering and still needs broth, stir in 1\nicefrac{1}{2}~cups from \textit{Medium Bowl~\#2} now. Refrigerate any remaining broth for another use.
\end{enumerate}

\subsection*{Curry}

\noindent
While the Instant Pot cooks: Drain \textbf{diced~pineapple} ---
Peel, seed, and cut 1 \textbf{butternut~squash} into 1" chunks ---
Peel and large-dice 4 \textbf{carrots} into \nicefrac{3}{4}--1" pieces ---
Large-dice 2 \textbf{white~onions}; mince 8 \textbf{garlic~cloves} and 1~Tbsp. \textbf{ginger} ---
Juice 1 \textbf{lime}; wedge the other ---
Chop 1~cup \textbf{cilantro}

\begin{enumerate}
    \item Heat 2~Tbsp. \textbf{coconut~oil} in a large Dutch oven over medium heat. Add \textbf{onions}, \textbf{garlic}, and \textbf{ginger} with \nicefrac{1}{4}~tsp. \textbf{kosher~salt} and cook, stirring often, until the \textbf{onions} are translucent and the aromatics are fragrant, about \textit{6--8~minutes}. Do not brown.

    \item Add 7~Tbsp. \textbf{Mae~Ploy~yellow~curry~paste} and cook, stirring constantly, until the paste darkens slightly and smells toasted and fragrant, about \textit{2--3~minutes}.

    \item Pour in about half of the first can of \textbf{coconut~cream} and stir until the paste is smooth. Add the remaining \textbf{coconut~cream} from both cans, 4~Tbsp. \textbf{brown~sugar}, 2~Tbsp. \textbf{fish~sauce}, and 2 bruised \textbf{lemongrass~stalks}. If \textit{Medium Bowl~\#2} is ready, stir in 1\nicefrac{1}{2}~cups \textbf{reserved~broth}; if the Instant Pot is still running, continue without broth and add it as soon as the broth is strained and skimmed. Bring to a simmer.

    \item Add \textbf{squash} and \textbf{carrots} and return to a gentle simmer. Cook uncovered, stirring occasionally, for about \textit{15~minutes} until both are tender: a paring knife inserted into the center of a chunk meets no resistance, and pieces hold their shape but are soft. The sauce should coat the back of a spoon. Continue simmering in \textit{2--3~minute} increments until the vegetables are tender. If the vegetables are tender before the \textbf{chicken} is ready, reduce heat to low and hold.

    \item Add \textbf{chicken}, \textbf{pineapple}, and 1~Tbsp. \textbf{dried~holy~basil}. Simmer for \textit{3--5~minutes} until the \textbf{chicken} is heated through and steaming hot throughout. 

    \item Remove the \textbf{lemongrass}. Stir in the juice of 1 \textbf{lime} and \nicefrac{1}{2}~cup of the chopped \textbf{cilantro}. Serve the curry over the fluffed \textbf{rice}, garnished with the remaining \textbf{cilantro} and \textbf{lime~wedges}. Taste and adjust seasoning with \textbf{salt} if needed.
\end{enumerate}

\subsection*{Equipment Required}
\begin{itemize}
    \item Rice cooker
    \item Instant Pot (6-quart)
    \item Large Dutch oven or heavy pot (6--7~quart capacity)
    \item Fine-mesh strainer (for washing rice and straining broth)
    \item Cutting board and chef's knife
    \item Medium bowls (2; for picked chicken and reserved broth)
    \item Measuring cups and spoons
    \item Wooden spoon or silicone spatula
    \item Rice paddle or fork
    \item Instant-read thermometer (for chicken)
    \item Tongs or slotted spoon
    \item Vegetable peeler
\end{itemize}

\subsection*{Mise en Place}
\begin{itemize}
    \item Before heat, portion shared aromatics: rice (\nicefrac{1}{2}~Tbsp. \textbf{coconut~oil}, 1 \textbf{bay~leaf}, 2 \textbf{cardamom~pods}); Instant Pot (1 \textbf{bay~leaf}, 3 \textbf{cardamom~pods}); curry (2~Tbsp. \textbf{coconut~oil})
    \item Portion \textbf{curry~paste}: 1~Tbsp. for Instant Pot liquid; 7~Tbsp. for the Dutch oven bloom
    \item \textit{Medium Bowl~\#1} --- picked \textbf{chicken}; \textit{Medium Bowl~\#2} --- 2~cups skimmed broth
    \item Hold half the \textbf{cilantro} and the \textbf{lime~wedges} for garnish (juice goes in the pot)
\end{itemize}

\subsection*{Ingredient Tips}
\begin{itemize}
    \item Mae Ploy \textbf{yellow~curry~paste} is recommended; 8~Tbsp. total is intentional and assertive---taste and adjust with \textbf{coconut~cream} or \textbf{brown~sugar} if needed
    \item \textbf{Coconut~cream} significantly improves body and richness; full-fat \textbf{coconut~milk} is an acceptable substitute but yields a thinner, less luxurious curry
    \item \textbf{Brown~sugar} (not palm) matches this tested balance with pineapple
    \item \textbf{Fish~sauce} brands vary in saltiness; start with 2~Tbsp. and adjust at the end
    \item \textbf{Dried~holy~basil} adds herbal lift when stirred in at the finish
    \item Optional \textbf{lemongrass}: bruise the stalks with the flat of a knife before adding to the curry; remove before serving
    \item Bone-in, skin-on \textbf{leg~quarters} yield richer Instant Pot broth; discard skin after cooking
\end{itemize}

\subsection*{Preparation Tips}
\begin{itemize}
    \item \textbf{Rice} usually needs \textit{3--4~rinses} to run clear; that keeps grains separate
    \item Oversized \textbf{squash} or \textbf{carrot} pieces lag; keep cuts uniform so they finish together
    \item The final simmer only reheats the \textbf{chicken}---avoid a hard boil or it shreds and dries
    \item Skim Instant Pot fat thoroughly; leftover fat muddies the coconut sauce
    \item If the sauce tastes flat at the end, add \textbf{lime~juice} before more \textbf{salt}
\end{itemize}

\subsection*{Make Ahead \& Storage}
\begin{itemize}
    \item Curry can be made \textit{1--2~days} ahead; flavors improve
    \item Store refrigerated up to \textit{4~days}
    \item Reheat gently; thin with reserved Instant Pot broth or \textbf{coconut~cream} if the sauce has thickened
    \item Add fresh \textbf{lime~juice} and \textbf{cilantro} after reheating
    \item Extra Instant Pot broth keeps refrigerated \textit{4--5~days} or freezes well
    \item Cooked \textbf{rice} keeps refrigerated \textit{3--4~days}; reheat with a splash of water
\end{itemize}

\subsection*{Serving Suggestions}
\begin{itemize}
    \item Serve the curry over the aromatic rice-cooker \textbf{rice}
    \item \textbf{Lime~wedges} and remaining \textbf{cilantro} at the table
    \item Fresh Thai basil at the table if available
\end{itemize}
