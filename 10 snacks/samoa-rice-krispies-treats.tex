\begin{quote}
\textit{Toasted unsweetened coconut and crisp rice cereal are bound with melted Werther's caramel and marshmallows. Squares are dipped and drizzled in dark chocolate and finished with flaky salt. Makes 24 bars.}
\end{quote}

\section*{Ingredients}
\setlength{\columnsep}{20pt}
\begin{multicols}{2}
\noindent
    Unsweetened coconut flakes \dotfill 1\nicefrac{1}{2} cups \\
    Salted butter, plus more for greasing \dotfill \nicefrac{1}{4} cup \\
    Crisp rice cereal \dotfill 6 cups \\
    Werther's chewy caramels \dotfill 1 cup \\
    Heavy cream \dotfill 1 Tbsp. \\
    Fine salt \dotfill \nicefrac{1}{4} tsp. \\
    \columnbreak
    Mini marshmallows \dotfill 5\nicefrac{1}{2} cups \\
    Mexican vanilla \dotfill 1 tsp. \\
    Dark chocolate chips (60--70\%) \dotfill 2 cups \\
    Coconut oil \dotfill 2 tsp. \\
    Flaky salt \dotfill 1 tsp. \\
\end{multicols}

\section*{Directions}

\noindent
Preheat oven to \textit{325°F} ---
Line a rimmed baking sheet with parchment ---
Line a \textit{9×13-inch} pan with parchment, leaving overhang on two long sides. Grease the parchment and \textit{Large Bowl~\#1} with \textbf{butter} ---
Measure 6~cups \textbf{crisp rice cereal} into \textit{Large Bowl~\#1} ---
Unwrap 1~cup \textbf{caramels} into \textit{Medium Bowl~\#1} ---
Measure 5\nicefrac{1}{2}~cups \textbf{mini marshmallows} and have ready ---
Combine 2~cups \textbf{dark chocolate chips} and 2~tsp. \textbf{coconut oil} in \textit{Medium Bowl~\#2} ---
Measure 1~tsp. \textbf{flaky salt} into \textit{Small Bowl~\#1}

\begin{enumerate}
    \item Toast the coconut:
    \begin{enumerate}
        \item Spread 1\nicefrac{1}{2}~cups \textbf{unsweetened coconut flakes} in an even layer on the parchment-lined sheet.
        \item Bake at \textit{325°F} for \textit{5--8~minutes}, stirring every \textit{2--3~minutes}, until most flakes are golden brown and the pan smells nutty. Pale centers are fine. Continue in \textit{1~minute} stretches until that color. Unsweetened flakes brown fast.
        \item Transfer the coconut at once to the \textbf{cereal} in \textit{Large Bowl~\#1} so it stops cooking on the hot pan. Toss until the coconut is mixed through the cereal.
    \end{enumerate}

    \item Melt the caramel marshmallow binder in a heavy 4--5~qt. pot:
    \begin{enumerate}
        \item Add \nicefrac{1}{4}~cup \textbf{butter} and 1~Tbsp. \textbf{heavy cream}. Melt over \textit{low} heat, stirring, until the butter is liquid and the cream is mixed in, about \textit{1--2~minutes}.
        \item Add the unwrapped \textbf{caramels} (\textit{Medium Bowl~\#1}). Cook over \textit{low} heat, stirring and scraping the bottom constantly, until the caramel is smooth and glossy and ribbons off the spatula, \textit{8--12~minutes}. Stiff or grainy caramel needs another teaspoon of \textbf{cream}. Stay on low and keep stirring. Scorched sugar on the bottom stays bitter.
        \item Stir in \nicefrac{1}{4}~tsp. \textbf{fine salt}. Taste. Add a pinch more salt until the caramel is salty-sweet.
        \item Add 5\nicefrac{1}{2}~cups \textbf{mini marshmallows}. Stir and fold over \textit{low} heat until just melted, pale tan, and stretchy, \textit{3--5~minutes}. A few small marshmallow streaks are fine. Pull the pot as soon as it is smooth. Brown marshmallow sets the bars hard.
        \item Take the pot off the heat. Stir in 1~tsp. \textbf{Mexican vanilla}.
    \end{enumerate}

    \item Pour the hot caramel marshmallow mixture over the \textbf{cereal} and \textbf{coconut} (\textit{Large Bowl~\#1}). Fold with a buttered spatula until every grain is coated and the coconut is evenly distributed, \textit{1--2~minutes}. Work fast. The binder thickens as it cools. Scrape the mixture into the prepared pan. Press gently into an even layer about 1~inch thick, using the buttered spatula or a sheet of buttered parchment. Fill the corners. Stop once the surface is level. Hard packing crushes the cereal and makes tough bars.

    \item Cool at room temperature until the slab feels set, the surface is no longer tacky-wet, and the center springs back slightly when pressed, \textit{45--60~minutes}. Continue cooling in \textit{15~minute} stretches until the middle holds that set.

    \item Coat with chocolate:
    \begin{enumerate}
        \item Use the parchment overhang to lift the slab onto a cutting board. Cut into 24 bars (6 columns × 4 rows). Wipe the knife clean between cuts.
        \item Cover the work surface with parchment. Microwave the \textbf{chocolate chips} and \textbf{coconut oil} (\textit{Medium Bowl~\#2}) in \textit{30-second} bursts, stirring after each, until smooth and fluid, about \textit{90--120~seconds} total. The chocolate should ribbon off a spoon, then settle. Thick chocolate needs another \textit{10~seconds} in the microwave. Matte, pasty chocolate needs another \nicefrac{1}{2}~tsp. \textbf{coconut oil}. Microwave \textit{10~seconds} and stir until it loosens.
        \item Dip the bottom of each bar in the chocolate. Let the excess drip back into the bowl. Set the bars chocolate-side down on the parchment.
        \item Drizzle the remaining chocolate over the tops in thin stripes. Sprinkle the \textbf{flaky salt} (\textit{Small Bowl~\#1}) while the drizzle is still wet.
    \end{enumerate}

    \item Let the chocolate set at room temperature until the coating is dry, firm, and peels cleanly from the parchment, \textit{20--30~minutes}. For a faster set, refrigerate \textit{10~minutes}, then return to room temperature. The bars are at their best once the chocolate is set and the interior is still chewy, about \textit{1~hour} after dipping. Store in an airtight container at room temperature.
\end{enumerate}

\section*{Hints and Notes}

\subsection*{Yield}
\begin{itemize}
    \item Makes 24 bars (6×4 grid from a 9×13 pan)
    \item Cut 12 larger squares for thicker handheld pieces
\end{itemize}

\subsection*{Mise en Place}
\begin{itemize}
    \item \textit{Medium Bowl~\#1} --- 1~cup unwrapped \textbf{Werther's Original chewy caramels} (about 20 pieces)
    \item \textit{Large Bowl~\#1} --- greased; 6~cups \textbf{crisp rice cereal}, with toasted \textbf{coconut} tossed in after the oven
    \item \textit{Medium Bowl~\#2} --- microwave-safe; 2~cups \textbf{dark chocolate chips} and 2~tsp. \textbf{coconut oil}
    \item \textit{Small Bowl~\#1} --- 1~tsp. \textbf{flaky salt}
    \item Measure 5\nicefrac{1}{2}~cups \textbf{mini marshmallows} and have ready
    \item Line the sheet and the 9×13 pan. Grease the pan parchment and \textit{Large Bowl~\#1} before the coconut goes in the oven
\end{itemize}

\subsection*{Ingredient Tips}
\begin{itemize}
    \item \textbf{Unsweetened coconut flakes} give chew and toasted flavor without extra sugar. Shredded coconut works at the same volume. Sweetened flakes make a much sweeter bar.
    \item \textbf{Werther's Original chewy caramels} melt into a butterscotch binder. One 5~oz. bag is about 21 pieces, close to 1~cup unwrapped. Soft caramel candies also work. Hard Werther's will not melt into a binder.
    \item \textbf{Mini marshmallows} (one 10~oz. bag is about 5\nicefrac{1}{2}~cups) melt evenly. Fresh marshmallows stay softer. Stale marshmallows make tough bars.
    \item \textbf{Crisp rice cereal}: a standard 12~oz. box holds about 12~cups, so this recipe uses half a box.
    \item \textbf{Dark chocolate} (60--70\%) cuts the caramel. Semisweet chips work if you want a sweeter coating.
    \item \textbf{Coconut oil} thins the chips so the bottoms dip cleanly. Neutral oil or shortening works in the same amount.
    \item \textbf{Heavy cream} helps the caramels liquefy. Whole milk works in the same amount.
\end{itemize}

\subsection*{Preparation Tips}
\begin{itemize}
    \item Stay on \textit{low} heat for the caramel and marshmallows. High heat scorches the sugar on the bottom and overcooks the marshmallows, which sets the bars hard.
    \item Butter the spatula and your hands before mixing and pressing. The binder grabs dry tools.
    \item Press just enough to level the pan. Hard packing crushes the cereal.
    \item Dip as soon as the bars hold their shape. Warm bars slump in the chocolate.
    \item Use a deep bowl for dipping so the chocolate stays in a pool. Leftover chocolate is expected.
    \item A zip-top bag with a snipped corner makes even drizzle stripes.
\end{itemize}

\subsection*{Make Ahead \& Storage}
\begin{itemize}
    \item Coconut can be toasted \textit{1~day} ahead. Cool and store airtight at room temperature.
    \item Unwrap the \textbf{caramels} ahead. They take time.
    \item Store bars in an airtight container at room temperature up to \textit{3~days}. They are best on day~1 and day~2.
    \item Refrigeration makes the bars hard. If you chill them, let them stand at room temperature \textit{15--20~minutes} before eating.
    \item Layer parchment between stacked bars so the chocolate does not stick.
    \item Freeze up to \textit{1~month}. Thaw in the container at room temperature.
\end{itemize}

\subsection*{Serving Suggestions}
\begin{itemize}
    \item Serve at cool room temperature, when the chocolate is firm and the center is still chewy.
    \item Pair with cold milk or black coffee.
    \item Cut into 1~inch bites for a dessert tray.
\end{itemize}
